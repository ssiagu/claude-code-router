- **覆盖率要求**: >90% (branches, functions, lines, statements)
- **测试分类**: 单元测试、集成测试、Mock测试
- **性能要求**: 完整测试套件执行时间 < 30秒

**关键测试场景**:
```typescript
describe('modeCommand Module', () => {
  describe('parseModelString', () => {
    it('should parse valid provider,model format', () => {
      expect(parseModelString('deepseek,deepseek-chat')).toEqual({
        provider: 'deepseek',
        model: 'deepseek-chat',
        original: 'deepseek,deepseek-chat'
      });
    });
    
    it('should return null for invalid formats', () => {
      expect(parseModelString('invalid')).toBeNull();
      expect(parseModelString('')).toBeNull();
      expect(parseModelString('provider,')).toBeNull();
      expect(parseModelString(',model')).toBeNull();
    });
  });
  
  describe('executeModeCommand', () => {
    beforeEach(() => {
      jest.clearAllMocks();
    });
    
    it('should show current mode when no arguments', async () => {
      mockReadConfigFile.mockResolvedValue(validConfig);
      const result = await executeModeCommand([]);
      expect(result.success).toBe(true);
      expect(result.message).toContain('当前模型');
    });
  });
});
```
